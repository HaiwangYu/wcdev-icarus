void OmnibusSigProc::restore_baseline(Array::array_xxf& arr)
{
    int nempty=0;
    int nnonfinite=0;
    for (int i = 0; i != arr.rows(); i++) {
        Waveform::realseq_t signal(arr.cols());
        int ncount = 0;
        for (int j = 0; j != arr.cols(); j++) {
            const float val = arr(i, j);
            if (val == 0) {
                continue;
            }
            // The 2D deconvolution (division by the response spectrum) can
            // produce NaN/Inf samples.  If they reach the baseline estimate
            // below, the median becomes NaN, the |x-baseline|<500 cut rejects
            // every sample and Waveform::median() is then called on an empty
            // vector, which throws "empty waveform" (see issue #491).  Drop
            // such samples here and zero them in-place so they neither poison
            // the baseline nor propagate downstream.
            if (!std::isfinite(val)) {
                ++nnonfinite;
                arr(i, j) = 0;
                continue;
            }
            signal.at(ncount) = val;
            ncount++;
        }
        if (!ncount) {
            ++nempty;
            continue;
        }
        signal.resize(ncount);
        float baseline = WireCell::Waveform::median(signal);

        Waveform::realseq_t temp_signal(ncount);
        int ncount2 = 0;
        for (size_t j = 0; j != signal.size(); j++) {
            if (fabs(signal.at(j) - baseline) < 500) {
                temp_signal.at(ncount2) = signal.at(j);
                ncount2++;
            }
        }
        temp_signal.resize(ncount2);

        // Guard against an empty second-pass selection: keep the first-pass
        // baseline rather than calling median() on an empty vector.
        if (ncount2) {
            baseline = WireCell::Waveform::median(temp_signal);
        }

        for (int j = 0; j != arr.cols(); j++) {
            if (arr(i, j) != 0) arr(i, j) -= baseline;
        }
    }
    if (nnonfinite) {
        log->warn("restore_baseline: zeroed {} non-finite (NaN/Inf) samples out of size=({},{})",
                  nnonfinite, arr.rows(), arr.cols());
    }
    if (nempty) {
        log->debug("{} empty rows out of size=({},{})",
                   nempty, arr.rows(), arr.cols());
    }
}
